# ── Runtime requirements — Meshtastic Terminal UI ─────────────────────────
# Python >= 3.9  (meshtastic constraint; tested on 3.11 / Pi OS Bookworm)

# ── Radio ──────────────────────────────────────────────────────────────────

# Meshtastic serial client — USB connection to Heltec LoRa V3
# Latest stable: 2.7.8  (2026-03)
meshtastic>=2.5.0,<3.0.0

# PubSub event bus — pinned to avoid breaking change in a future major
PyPubSub>=4.0.3,<5.0.0

# Serial port — meshtastic pins <4.0; no newer stable exists
pyserial>=3.5,<4.0

# Protocol Buffers — meshtastic uses proto3
protobuf>=4.21.12,<6.0.0

# HTTP client — used by meshtastic internally
requests>=2.31.0,<3.0.0

# YAML — meshtastic config serialisation
PyYAML>=6.0.1,<7.0.0

# ── Optional hardware peripherals (GPIO) ──────────────────────────────────
# gpiozero: high-level GPIO API for Pi.
#   Provides: RotaryEncoder, Button, TonalBuzzer, Buzzer
#   Requires a pin factory backend on the Pi:
#     sudo apt-get install -y python3-gpiozero python3-lgpio
#   On non-Pi (dev desktop) gpiozero raises BadPinsError on init —
#   the hardware manager catches this and runs in stub mode.
gpiozero>=2.0,<3.0

# ── Optional GPS ──────────────────────────────────────────────────────────
# NMEA 0183 sentence parser (GPS serial module)
pynmea2>=1.19.0,<2.0.0
