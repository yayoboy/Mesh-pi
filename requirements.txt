# ── Runtime requirements — Meshtastic Terminal UI ─────────────────────────
# Python >= 3.9  (meshtastic constraint; tested on 3.11 / Pi OS Bookworm)

# ── Radio ──────────────────────────────────────────────────────────────────

# Meshtastic serial client — USB connection to Heltec LoRa V3
# Latest stable: 2.7.8  (2026-03)
meshtastic>=2.5.0,<3.0.0

# PubSub event bus — pinned to avoid breaking change in a future major
PyPubSub>=4.0.3,<5.0.0

# Serial port — meshtastic pins <4.0; no newer stable exists
pyserial>=3.5,<4.0

# Protocol Buffers — meshtastic uses proto3
protobuf>=4.21.12,<6.0.0

# HTTP client — used by meshtastic internally
requests>=2.31.0,<3.0.0

# YAML — meshtastic config serialisation
PyYAML>=6.0.1,<7.0.0

# ── Optional hardware peripherals (GPIO) ──────────────────────────────────
# gpiozero: high-level GPIO API for Pi.
#   Provides: RotaryEncoder, Button, TonalBuzzer, Buzzer
#   Requires a pin factory backend on the Pi:
#     sudo apt-get install -y python3-gpiozero python3-lgpio
#   On non-Pi (dev desktop) gpiozero raises BadPinsError on init —
#   the hardware manager catches this and runs in stub mode.
gpiozero>=2.0,<3.0

# ── Optional GPS ──────────────────────────────────────────────────────────
# NMEA 0183 sentence parser (GPS serial module)
pynmea2>=1.19.0,<2.0.0

# ── Optional I2C telemetry sensors ────────────────────────────────────────
# Low-level I2C bus access — used by SHT30 driver and as a backend for
# RPi.bme280. Also pulled in transitively by pi-ina219.
smbus2>=0.4.2,<1.0.0

# INA219 / INA226 voltage, current and power monitor
# Handles sensor calibration registers and fixed-point arithmetic.
pi-ina219>=1.4.1,<2.0.0

# BME280 temperature + humidity + pressure sensor
# Thin wrapper around smbus2 that handles the BME280 compensation formulas.
RPi.bme280>=0.2.4,<1.0.0

# ── Notes ─────────────────────────────────────────────────────────────────
# tkinter must be installed as a system package, NOT via pip:
#   sudo apt-get install -y python3-tk
#
# bleak (BLE) is a hard dependency of meshtastic even for USB-only setups.
# It requires D-Bus on Linux:
#   sudo apt-get install -y python3-dbus libglib2.0-dev
#
# gpiozero on Pi OS Bookworm uses lgpio as its default pin factory:
#   sudo apt-get install -y python3-lgpio
#
# I2C must be enabled on the Raspberry Pi:
#   sudo raspi-config → Interface Options → I2C → Enable
# or add "dtparam=i2c_arm=on" to /boot/firmware/config.txt and reboot.
#
# Verify I2C devices with: i2cdetect -y 1
#   sudo apt-get install -y i2c-tools
